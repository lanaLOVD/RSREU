.model small
.stack 100h

.data
num dw 25

.code

; ===== макрос установки курсора =====
SetCursor MACRO row,col
    push ax
    push bx
    push dx

    mov ah,02h
    mov bh,0
    mov dh,row
    mov dl,col
    int 10h

    pop dx
    pop bx
    pop ax
ENDM


; ===== макрос вывода двоичного числа =====
PrintBin MACRO value,row,col
LOCAL L1,ONE

    push ax
    push bx
    push cx
    push dx

    SetCursor row,col

    mov bx,value
    mov cx,16

L1:
    shl bx,1
    jc ONE

    mov dl,'0'
    jmp PRINT

ONE:
    mov dl,'1'

PRINT:
    mov ah,02h
    int 21h
    loop L1

    pop dx
    pop cx
    pop bx
    pop ax
ENDM


start:
    mov ax,@data
    mov ds,ax

    PrintBin num,10,30

    mov ax,4C00h
    int 21h

end start
