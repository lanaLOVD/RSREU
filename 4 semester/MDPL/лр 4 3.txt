.model small
.stack 100h

.data
msg db 'Vector F: $'

x1 db ?
x2 db ?
x3 db ?

t1 db ?
t2 db ?
t3 db ?
t4 db ?
t5 db ?
t6 db ?

.code
start:
    mov ax,@data
    mov ds,ax

    mov dx,offset msg
    mov ah,09h
    int 21h

    mov cx,8
    xor bx,bx

main_loop:

; ----- получаем X1 X2 X3 -----

    mov al,bl
    and al,4
    shr al,2
    mov x1,al

    mov al,bl
    and al,2
    shr al,1
    mov x2,al

    mov al,bl
    and al,1
    mov x3,al

; ----- X1∧X2 -----

    mov al,x1
    and al,x2
    mov t1,al

; ----- X2∧X3 -----

    mov al,x2
    and al,x3
    mov t2,al

; ----- NOR -----
; X1∧X2 ↓ X2∧X3

    mov al,t1
    or al,t2
    not al
    and al,1
    mov t3,al

; ----- X1∧X3 -----

    mov al,x1
    and al,x3
    mov t4,al

; ----- ¬X1 -----

    mov al,x1
    xor al,1
    mov t5,al

; ----- XOR -----

    mov al,t4
    xor al,t5
    mov t6,al

; ----- итоговая NOR -----

    mov al,t3
    or al,t6
    not al
    and al,1

; ----- вывод -----

    add al,'0'
    mov dl,al
    mov ah,02h
    int 21h

    inc bx
    loop main_loop

exit:
    mov ax,4C00h
    int 21h

end start
