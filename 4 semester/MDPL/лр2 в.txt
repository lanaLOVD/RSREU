.model small
.stack 100h

.data
a dw 10
b dw 3
c dw 4
d dw 2
w dw 7
f dw 3

z dw ?
N dw ?

msgZ db 'z = $'
msgN db 13,10,'N = $'

.code
start:
    mov ax, @data
    mov ds, ax

; ===== z = w - c - f =====
    mov ax, w
    sub ax, c
    sub ax, f
    mov z, ax

    cmp ax, 0
    jne else_part

; ===== ЕСЛИ z = 0 =====
    mov ax, a
    add ax, b

    mov bx, f
    imul bx

    mov bx, a
    sub bx, b

    cwd
    idiv bx

    add ax, a
    add ax, f

    mov N, ax
    jmp print_result

; ===== ИНАЧЕ =====
else_part:
    mov ax, a
    add ax, b
    add ax, d

    mov bx, w
    imul bx

    mov bx, a
    add bx, w

    cwd
    idiv bx

    mov bx, c
    sub bx, ax

    mov N, bx

; ===== Вывод =====
print_result:

    ; вывод z
    mov dx, offset msgZ
    mov ah, 09h
    int 21h

    mov ax, z
    call print_num

    ; вывод N
    mov dx, offset msgN
    mov ah, 09h
    int 21h

    mov ax, N
    call print_num

finish:
    mov ax, 4C00h
    int 21h

; ===== Процедура вывода числа =====
print_num proc
    push ax
    push bx
    push cx
    push dx

    xor cx, cx
    mov bx, 10

convert:
    xor dx, dx
    div bx
    push dx
    inc cx
    cmp ax, 0
    jne convert

print_loop:
    pop dx
    add dl, '0'
    mov ah, 02h
    int 21h
    loop print_loop

    pop dx
    pop cx
    pop bx
    pop ax
    ret
print_num endp

end start
