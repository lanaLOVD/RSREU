.model small
.stack 100h

.data
num dw 25        ; число которое будем выводить

.code

; ===== макрос установки курсора =====
SetCursor MACRO row,col
    push ax
    push bx
    push dx

    mov ah,02h
    mov bh,0
    mov dh,row
    mov dl,col
    int 10h

    pop dx
    pop bx
    pop ax
ENDM

; ===== макрос вывода двоичного числа =====
PrintBin MACRO value,row,col
LOCAL loop1

    push ax
    push bx
    push cx
    push dx

    SetCursor row,col

    mov ax,value
    mov cx,16

loop1:
    shl ax,1
    jc one

    mov dl,'0'
    jmp print

one:
    mov dl,'1'

print:
    mov ah,02h
    int 21h
    loop loop1

    pop dx
    pop cx
    pop bx
    pop ax
ENDM

start:

    mov ax,@data
    mov ds,ax

    PrintBin num,10,30   ; вывод числа в позицию

    mov ax,4C00h
    int 21h

end start
