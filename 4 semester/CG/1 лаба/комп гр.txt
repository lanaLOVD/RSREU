using System;
using System.Drawing;
using System.Windows.Forms;

namespace Spiral19
{
    public partial class Form1 : Form
    {
        private const int PointCount = 300000;

        // Вероятности
        private readonly float[] probability =
        {
            0.3f,
            0.7f
        };

        // Коэффициенты (a, b, c, d, e, f)
        private readonly float[,] func =
        {
            { 0.2f, -0.3f,  0.3f, 0.2f, 0.0f, 0.0f },
            { 0.5f,  0.7f, -0.7f, 0.5f, 0.5f, 0.7f }
        };

        private Bitmap bmp;
        private Random rnd = new Random();

        public Form1()
        {
            InitializeComponent();
            DrawFractal();
        }

        private void DrawFractal()
        {
            bmp = new Bitmap(pictureBox1.Width, pictureBox1.Height);

            float x = 0, y = 0;

            for (int i = 0; i < PointCount; i++)
            {
                double r = rnd.NextDouble();
                int index = r < probability[0] ? 0 : 1;

                float newX = func[index, 0] * x + func[index, 1] * y + func[index, 4];
                float newY = func[index, 2] * x + func[index, 3] * y + func[index, 5];

                x = newX;
                y = newY;

                int px = (int)(x * 100) + pictureBox1.Width / 2;
                int py = pictureBox1.Height - (int)(y * 100);

                if (px >= 0 && px < pictureBox1.Width &&
                    py >= 0 && py < pictureBox1.Height)
                {
                    bmp.SetPixel(px, py, Color.Cyan);
                }
            }

            pictureBox1.Image = bmp;
        }
    }
}
