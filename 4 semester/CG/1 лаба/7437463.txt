using System;
using System.Drawing;
using System.Windows.Forms;
using System.Collections.Generic;

namespace Spiral19
{
    public partial class Form1 : Form
    {
        private const int PointNumber = 400000;

        private float[] _probability = { 0.3f, 0.7f };

        private float[,] _funcCoef =
        {
            { 0.2f, -0.3f,  0.3f, 0.2f, 0.0f, 0.0f },
            { 0.5f,  0.7f, -0.7f, 0.5f, 0.5f, 0.7f }
        };

        public Form1()
        {
            InitializeComponent();
            DrawFractal();
        }

        private void DrawFractal()
        {
            Bitmap bmp = new Bitmap(pictureBox1.Width, pictureBox1.Height);
            Graphics g = Graphics.FromImage(bmp);
            g.Clear(Color.Black);

            Random rnd = new Random();

            float x = 0, y = 0;

            List<PointF> points = new List<PointF>();

            float minX = float.MaxValue, maxX = float.MinValue;
            float minY = float.MaxValue, maxY = float.MinValue;

            //  Генерируем точки и ищем реальные границы
            for (int i = 0; i < PointNumber; i++)
            {
                int index = rnd.NextDouble() <= _probability[0] ? 0 : 1;

                float newX =
                    _funcCoef[index, 0] * x +
                    _funcCoef[index, 1] * y +
                    _funcCoef[index, 4];

                float newY =
                    _funcCoef[index, 2] * x +
                    _funcCoef[index, 3] * y +
                    _funcCoef[index, 5];

                x = newX;
                y = newY;

                points.Add(new PointF(x, y));

                if (x < minX) minX = x;
                if (x > maxX) maxX = x;
                if (y < minY) minY = y;
                if (y > maxY) maxY = y;
            }

            float scaleX = pictureBox1.Width / (maxX - minX);
            float scaleY = pictureBox1.Height / (maxY - minY);
            float scale = Math.Min(scaleX, scaleY) * 0.9f; // небольшой отступ

            //  Рисуем уже масштабированные точки
            foreach (var p in points)
            {
                int px = (int)((p.X - minX) * scale) + 20;
                int py = pictureBox1.Height - (int)((p.Y - minY) * scale) - 20;

                if (px >= 0 && px < pictureBox1.Width &&
                    py >= 0 && py < pictureBox1.Height)
                {
                    bmp.SetPixel(px, py, Color.Cyan);
                }
            }

            pictureBox1.Image = bmp;
        }
    }
}
